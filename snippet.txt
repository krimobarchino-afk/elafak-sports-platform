function syncParticipationDate(){let aid=document.getElementById('partActivity')?.value||'';let a=db.activities.find(x=>x.id===aid);let b=document.getElementById('partActivityDateBox');if(b)b.textContent=a?'تاريخ المشاركة مرتبط بالنشاط: '+formatActivitySchedule(a):'التاريخ يُؤخذ تلقائيًا من ملف النشاط.'}
function save(){localStorage.setItem(KEY,JSON.stringify(db));localStorage.setItem('afak_settings_v15',JSON.stringify(settings));localStorage.setItem('afak_packs_v14',JSON.stringify(packs));}
function clearForm(ids){ids.forEach(id=>{let e=document.getElementById(id);if(!e)return;if(e.type==='file')e.value='';else if(e.tagName==='SELECT')e.selectedIndex=0;else e.value='';});}
function nextId(prefix,arr){let nums=arr.map(x=>parseInt(String(x.id||'').replace(/\D/g,''),10)).filter(Number.isFinite);return prefix+String((nums.length?Math.max(...nums):0)+1).padStart(4,'0');}
function show(id){document.querySelectorAll('.section').forEach(x=>x.classList.remove('active'));document.getElementById(id).classList.add('active');render();scrollTo(0,0)}
function toggle(id){document.getElementById(id).classList.toggle('hide')}
function fillSelect(id,arr){let el=document.getElementById(id);if(el)el.innerHTML=arr.map(x=>`<option>${x}</option>`).join('')}
